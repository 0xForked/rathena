//===== eXRo Script ==========================================
//= Chaos Hour + Double-EXP Weekend timers  (PLAN-017 / Phase C)
//===== Description: =========================================
//= Two schedule-driven toggles. No conf/import edit — the
//= battle-flag change is a runtime setbattleflag, and the map
//= flag lives here.
//=
//=   Chaos Hour — mf_pvp ON for one hour on ONE dedicated
//=     arena map (guild_vs3: no warps lead there, nobody
//=     levels on it), with a 10-minute pre-announce. Never a
//=     town, never a levelling map, so nobody is stranded in
//=     PvP. setmapflag / removemapflag are idempotent.
//=
//=   Double-EXP Weekend — Fri 00:00 -> Mon 00:00 UTC, base &
//=     job exp x2 via setbattleflag. setbattleflag does NOT
//=     survive a restart, so OnInit re-applies it if the box
//=     reboots inside the window.
//=
//= ponytail: the x1 baseline 3800 is hard-coded in one .const
//= here — keep it in sync with tools/docker/asset/battle_conf.txt
//= and www/lib/config/rates.ts (same cross-link as rates.ts).
//= Container clock is UTC (PLAN-012); every OnClock time is UTC.
//============================================================

-	script	exro_timers	-1,{
	end;

OnInit:
	.chaos_map$ = "guild_vs3";
	.base_x1 = 3800;      // PLAN-015 baseline (38x). Mirror: asset/battle_conf.txt + www/lib/config/rates.ts
	.base_x2 = 7600;      // 2x the baseline

	// re-apply the weekend multiplier if this restart landed inside the window.
	// DT_DAYOFWEEK: 0=Sun .. 6=Sat. Window = Fri(5), Sat(6), Sun(0); ends Mon 00:00.
	.@d = gettime(DT_DAYOFWEEK);
	if (.@d == 5 || .@d == 6 || .@d == 0) {
		setbattleflag "base_exp_rate", .base_x2;
		setbattleflag "job_exp_rate", .base_x2;
		debugmes "exro_timers: restart inside Double-EXP weekend — re-applied x2.";
	}
	end;

// ---- Chaos Hour: 19:00-20:00 UTC on .chaos_map$ ----------
OnClock1850:
	announce "[eXRo] Chaos Hour on " + .chaos_map$ + " in 10 minutes — free-for-all PvP on that map.", bc_all;
	end;

OnClock1900:
	setmapflag .chaos_map$, mf_pvp;
	announce "[eXRo] Chaos Hour is LIVE — PvP is ON on " + .chaos_map$ + " for one hour.", bc_all;
	end;

OnClock2000:
	removemapflag .chaos_map$, mf_pvp;
	announce "[eXRo] Chaos Hour is over — PvP on " + .chaos_map$ + " is off.", bc_all;
	end;

// ---- Double-EXP Weekend: Fri 00:00 -> Mon 00:00 UTC ------
OnFri0000:
	setbattleflag "base_exp_rate", .base_x2;
	setbattleflag "job_exp_rate", .base_x2;
	announce "[eXRo] Double-EXP Weekend has begun! Base & Job EXP are x2 until Monday 00:00 UTC.", bc_all;
	end;

OnMon0000:
	setbattleflag "base_exp_rate", .base_x1;
	setbattleflag "job_exp_rate", .base_x1;
	announce "[eXRo] Double-EXP Weekend has ended — EXP rates are back to normal.", bc_all;
	end;
}
