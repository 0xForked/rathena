//===== SeROja Script ==========================================
//= Equipment Mall + client bridge  (PLAN-018 Phase D / D4, D4c)
//===== Description: =========================================
//= A curated ZENY SINK over the www-owned www_mall_catalog table
//= (query_sql, ADR-0003). Two front ends, one buy path:
//=
//=   * `Equipment Mall` NPC (prontera,162,185) — clickable menu.
//=   * `seroja_mall_bridge` floating NPC — `bindatcmd "exml"`. The
//=     SeROja client window (src/UI/Components/SeROjaMall) sends
//=     "@exml <nonce> <op> <args>" and reads the dispbottom TSV
//=     reply (SEROJA<TAB><nonce><TAB>ROW|END|ERR<TAB>...), same
//=     protocol as market_client.txt.
//=
//= Money-safe ordering (same direction as zenysinks.txt): Zeny is
//= debited on the CHARACTER first, THEN getitem (RODEX if the
//= stack would overweight). A crash between costs the buyer the
//= zeny, never a dupe.
//=
//= job_mask (one INT on the catalog row): 0 -> everyone; else a
//= bitmask over the six 1st-job archetypes matched against BaseJob
//= (Swordman 1 / Mage 2 / Archer 4 / Acolyte 8 / Merchant 16 /
//= Thief 32; advanced classes collapse to their root; Novice /
//= SNovice / Taekwon / Doram = 0, see only job_mask = 0 rows).
//===== Column contract (must match www/db/schema.sql) =======
//= www_mall_catalog: id, item_id, category, price_zeny (BIGINT),
//=   job_mask (INT), qty_per_buy, enabled (TINYINT), sort (INT)
//============================================================

// buyer job archetype bit for the current (attached) player
function	script	F_SeROjaJobBit	{
	return
		(BaseJob == Job_Swordman) ? 1 :
		(BaseJob == Job_Mage)     ? 2 :
		(BaseJob == Job_Archer)   ? 4 :
		(BaseJob == Job_Acolyte)  ? 8 :
		(BaseJob == Job_Merchant) ? 16 :
		(BaseJob == Job_Thief)    ? 32 : 0;
}

// Shared buy path. Returns "OK" or a short reason. mes-free.
// args: getarg(0) catalog row id, getarg(1) pack count (1..99), getarg(2) job bit
function	script	F_SeROjaMallBuy	{
	.@row_id = getarg(0);
	.@packs  = getarg(1);
	.@bit    = getarg(2);
	if (.@packs < 1 || .@packs > 99)
		return "bad quantity";

	.@n = query_sql(
		"SELECT item_id, price_zeny, qty_per_buy FROM `www_mall_catalog` "
		+ "WHERE id = " + .@row_id + " AND enabled = 1 "
		+ "AND (job_mask = 0" + ((.@bit) ? " OR (job_mask & " + .@bit + ") <> 0" : "") + ")",
		.@iid, .@pz, .@qpb);
	if (.@n < 1)
		return "not available";
	if (getitemname(.@iid[0]) == "null")
		return "catalog item missing";

	.@cost  = .@pz[0] * .@packs;
	.@total = .@qpb[0] * .@packs;
	if (Zeny < .@cost)
		return "need " + .@cost + "z";

	Zeny -= .@cost;   // debit FIRST
	if (checkweight(.@iid[0], .@total)) {
		getitem .@iid[0], .@total;
	} else {
		setarray .@m_id[0], .@iid[0];
		setarray .@m_amt[0], .@total;
		mail getcharid(0), "SeROja Mall", "Equipment Mall order",
			"Too heavy to hand over in person — here's your " + getitemname(.@iid[0]) + ".",
			0, .@m_id, .@m_amt;
	}
	return "OK";
}

// ---------------------------------------------------------------------------
// Clickable NPC
// ---------------------------------------------------------------------------
prontera,162,185,4	script	Equipment Mall	4_F_04,{
	mes "[Equipment Mall]";
	mes "Curated gear and supplies, priced in zeny.";
	next;
	.@bit = callfunc("F_SeROjaJobBit");
	.@jobclause$ = " AND (job_mask = 0" + ((.@bit) ? " OR (job_mask & " + .@bit + ") <> 0" : "") + ")";

	.@cat$ = "";
	switch (select("Everything", "Weapons", "Armor", "Cards", "Consumables", "Costumes", "Etc", "Leave")) {
	case 2: .@cat$ = "weapon"; break;
	case 3: .@cat$ = "armor"; break;
	case 4: .@cat$ = "card"; break;
	case 5: .@cat$ = "consumable"; break;
	case 6: .@cat$ = "costume"; break;
	case 7: .@cat$ = "etc"; break;
	case 8: close;
	}

	.@page = 0;
	while (1) {
		.@where$ = "enabled = 1" + .@jobclause$;
		if (.@cat$ != "") .@where$ += " AND category = '" + escape_sql(.@cat$) + "'";
		.@n = query_sql(
			"SELECT id, item_id, price_zeny, qty_per_buy FROM `www_mall_catalog` "
			+ "WHERE " + .@where$ + " ORDER BY sort ASC, id ASC LIMIT 8 OFFSET " + (.@page * 8),
			.@row_id, .@iid, .@pz, .@qpb);
		if (.@n < 1) {
			mes "[Equipment Mall]";
			mes((.@page == 0) ? "Nothing in that section for your class yet." : "No more items.");
			close;
		}
		.@menu$ = "";
		for (.@i = 0; .@i < .@n; ++.@i) {
			.@nm$ = getitemname(.@iid[.@i]);
			if (.@nm$ == "null") .@nm$ = "Item #" + .@iid[.@i] + " (missing)";
			.@menu$ += .@nm$ + " x" + .@qpb[.@i] + " - " + .@pz[.@i] + "z:";
		}
		.@menu$ += "> Next page:< Back";
		.@s = select(.@menu$);
		if (.@s == .@n + 1) { ++.@page; continue; }
		if (.@s == .@n + 2) { if (.@page == 0) close; --.@page; continue; }
		.@p = .@s - 1;

		mes "[Equipment Mall]";
		mes "^0055FF" + getitemname(.@iid[.@p]) + "^000000";
		mes "Pack size: " + .@qpb[.@p] + "   Price: ^0055FF" + .@pz[.@p] + "z^000000 per pack";
		mes "How many packs?";
		next;
		input .@packs, 0, 99;
		if (.@packs < 1) { mes "[Equipment Mall]"; mes "Cancelled."; close; }

		.@r$ = callfunc("F_SeROjaMallBuy", .@row_id[.@p], .@packs, .@bit);
		mes "[Equipment Mall]";
		if (.@r$ == "OK") mes "Sold. Check your pack" + ((checkweight(.@iid[.@p], 1)) ? "." : " / RODEX.");
		else mes "Couldn't sell that: ^777777" + .@r$ + "^000000.";
		close;
	}
}

// ---------------------------------------------------------------------------
// Client bridge — bindatcmd "exml".
// params: [0] nonce  [1] op  [2..] op args
// ---------------------------------------------------------------------------
-	script	seroja_mall_bridge	-1,{
	end;

OnInit:
	bindatcmd "exml", strnpcinfo(3) + "::OnCmd", 0, 99;
	end;

OnCmd:
	if (.@atcmd_numparameters < 2) {
		dispbottom "SEROJA\t0\tERR\tbad request";
		end;
	}
	.@nonce$ = .@atcmd_parameters$[0];
	.@op$    = .@atcmd_parameters$[1];
	.@bit    = callfunc("F_SeROjaJobBit");

	if (.@op$ == "browse") {
		.@cat$  = .@atcmd_parameters$[2];
		.@page  = atoi(.@atcmd_parameters$[3]);
		.@where$ = "enabled = 1 AND (job_mask = 0"
			+ ((.@bit) ? " OR (job_mask & " + .@bit + ") <> 0" : "") + ")";
		if (.@cat$ != "" && .@cat$ != "-")
			.@where$ += " AND category = '" + escape_sql(.@cat$) + "'";
		.@n = query_sql(
			"SELECT id, item_id, price_zeny, qty_per_buy FROM `www_mall_catalog` "
			+ "WHERE " + .@where$ + " ORDER BY sort ASC, id ASC LIMIT 8 OFFSET " + (.@page * 8),
			.@id, .@iid, .@pz, .@qpb);
		for (.@i = 0; .@i < .@n; ++.@i) {
			.@nm$ = getitemname(.@iid[.@i]);
			if (.@nm$ == "null") .@nm$ = "Item #" + .@iid[.@i];
			dispbottom "SEROJA\t" + .@nonce$ + "\tROW\t" + .@id[.@i] + "\t" + .@nm$
				+ "\t" + .@qpb[.@i] + "\t" + .@pz[.@i];
		}
		dispbottom "SEROJA\t" + .@nonce$ + "\tEND\tbrowse";
		end;
	}

	if (.@op$ == "buy") {
		.@row_id = atoi(.@atcmd_parameters$[2]);
		.@packs  = atoi(.@atcmd_parameters$[3]);
		.@r$ = callfunc("F_SeROjaMallBuy", .@row_id, .@packs, .@bit);
		if (.@r$ == "OK") dispbottom "SEROJA\t" + .@nonce$ + "\tEND\tbuy";
		else              dispbottom "SEROJA\t" + .@nonce$ + "\tERR\t" + .@r$;
		end;
	}

	dispbottom "SEROJA\t" + .@nonce$ + "\tERR\tunknown op";
	end;
}
