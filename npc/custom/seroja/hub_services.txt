//===== SeROja Script ==========================================
//= Hub Services — storage + mount rental at the Prontera plaza
//===== Description: =========================================
//= RPT-004 #16 + #17 (+ helps #19). Two small QoL NPCs at the
//= Prontera plaza hub. Storage Keeper stays at its original spot;
//= Rental Master moved to 172,178 (east side, clear of the
//= fountain-front stock NPC cluster) since its original tile was
//= hard to click apart from the crowd -- confirmed walkable and
//= reachable in-client. A wider hub reshuffle was attempted and
//= reverted: some of the new coordinates picked (e.g. Daily Hunt's)
//= turned out to be off the walkable plaza ("Invalid coordinates"
//= on warp) -- reverting untested guesses beats shipping a broken
//= NPC. Re-attempt de-crowding only with coordinates verified
//= walkable first.
//=
//= #16 Storage Keeper — opens account storage. `openstorage`
//= needs no Kafra card / fee here; a convenience desk, the real
//= Kafra network still exists for save/warp/cart.
//=
//= #17 Rental Master — free mounts.
//=  - "Rental Mount" = `setmounting()` (SC_ALL_RIDING). Works for
//=    EVERY class, no class skill needed — this is the reliable
//=    one and also covers a Crusader who never learned KN_RIDING
//=    (#19: the stock Peco Peco Breeder at prontera,232,318 still
//=    exists for the real Peco once KN_RIDING is learned).
//=  - Cart / Falcon / Peco / Dragon = the class-native mounts;
//=    the engine only grants these if the player's job has the
//=    matching skill (MC_PUSHCART / HT_FALCON / KN_RIDING /
//=    RK_DRAGONTRAINING), otherwise the pick is a no-op.
//= All toggles last until dismissed here or a job change / relog
//= recalc. ponytail: a timed rentitem mount is the upgrade path.
//============================================================

prontera,160,188,4	script	Storage Keeper	4_F_KAFRA1,{
	mes "[Storage Keeper]";
	mes "Need your storage? Right here.";
	next;
	if (select("Open storage", "Leave") == 2)
		close;
	close2;
	openstorage;
	end;
}

prontera,172,178,4	script	Rental Master	4_M_JOB_KNIGHT2,{
	mes "[Rental Master]";
	mes "Free mounts. 'Rental Mount' works for any class; the others need your job's mount skill.";
	next;
	switch (select("Rental Mount (any class)", "Cart", "Falcon", "Peco Peco", "Dragon", "Dismiss all", "Leave")) {
	case 1:
		if (ismounting()) {
			mes "[Rental Master]";
			mes "You're already on a rental mount.";
			close;
		}
		setmounting();
		break;
	case 2: setcart 1; break;
	case 3: setfalcon 1; break;
	case 4: setriding 1; break;
	case 5: setdragon 1; break;
	case 6:
		if (ismounting())
			setmounting();
		setcart 0;
		setfalcon 0;
		setriding 0;
		setdragon 0;
		setmadogear 0;
		mes "[Rental Master]";
		mes "All dismissed.";
		close;
	case 7:
		close;
	}
	mes "[Rental Master]";
	mes "There you go. Come back any time to dismiss it.";
	close;
}
