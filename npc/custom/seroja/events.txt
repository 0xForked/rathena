//===== SeROja Script ==========================================
//= Event Framework + Custom Invasions  (PLAN-017 / Phase C)
//===== Description: =========================================
//= A reusable core for scheduled events: an hourly rotation,
//= a per-player kill tally, a top-N ranking, and one payout
//= helper (F_EventReward) that every SeROja event routes its
//= rewards through.
//=
//=   seroja_events  — floating -1 controller.
//=     OnInit        builds the rotation + data tables.
//=     OnMinute55    5-min pre-announce of the NEXT slot.
//=     OnMinute00    advances .slot, starts that slot's event.
//=     OnStartTown / OnStartBoss / OnStopEvent  — GM overrides,
//=                   also reachable via the @event atcommand.
//=
//=   Tally: invasion mobs carry the kill label OnInvKill. The
//=   killer is appended to the global-temp parallel arrays
//=   $@ev_aid[] / $@ev_cid[] / $@ev_name$[] / $@ev_score[] and
//=   its score bumped. Cleared at each event START, so a crash
//=   mid-event just abandons that round. This ranks KILL COUNT,
//=   not damage — a leech that never lands a kill scores 0.
//=   ponytail: damage-weighted needs a per-hit hook; not worth
//=   it for a town invasion. Revisit only if players complain.
//=
//=   Rewards: F_EventReward(charid, accountid, tier) mails the
//=   tier's zeny + items (so offline winners still get paid),
//=   then — only if the winner is still online — grants
//=   #SEROJACOIN / #ACTIVITYCOIN via the PLAN-016 helpers. A
//=   winner who logs off between event end and payout loses
//=   just the coin portion. Announces say "rewards mailed".
//=
//= Container clock is UTC (PLAN-012); every announce time is
//= UTC. Zero www / schema change (PLAN-017 decision D3).
//= Item ids reuse the set verified in currency.txt (2026-09-10).
//============================================================

-	script	seroja_events	-1,{
	end;

OnInit:
	// hourly rotation — TOWN at slots 0 & 3, BOSS at 1 & 5, quiet otherwise
	setarray .rotation$[0], "TOWN","BOSS","IDLE","TOWN","IDLE","BOSS";
	.rot_size = getarraysize(.rotation$);
	.slot = -1;                       // first OnMinute00 advances to slot 0

	// Town Invasion — rotating town, plaza-ish spawn coords (not the warp tile)
	setarray .inv_towns$[0], "prontera","geffen","payon","aldebaran","morocc";
	setarray .inv_x[0],       155,        119,     179,    140,        156;
	setarray .inv_y[0],       178,        66,      106,    110,        93;
	.inv_wave_mob = 1023;             // Orc Warrior
	.inv_wave_qty = 20;
	.inv_boss_mob = 1087;            // Orc Hero (mini-boss)
	.inv_minutes  = 15;             // NOTE: the real value lives in the OnTimer900000 label
	.inv_topn     = 6;
	.inv_name$    = "SeROja Marauder";

	// Roaming Boss — wandering mini-MVP on a random field map
	setarray .boss_maps$[0], "prt_fild01","gef_fild00","pay_fild04","mjolnir_02","moc_fild02";
	.boss_mob       = 1115;          // Eddga
	.boss_name$     = "SeROja Wandering Terror";
	.boss_despawn_s = 3600;

	bindatcmd "event", "seroja_events::OnGMEvent", 99, 99;
	end;

// ---- hourly scheduler --------------------------------------
OnMinute55:
	.@next$ = .rotation$[(.slot + 1) % .rot_size];
	if (.@next$ == "TOWN")
		announce "[SeROja] A town is about to come under siege — be ready in 5 minutes.", bc_all;
	else if (.@next$ == "BOSS")
		announce "[SeROja] A roaming terror stirs — a sighting in 5 minutes.", bc_all;
	end;

OnMinute00:
	.slot = (.slot + 1) % .rot_size;
	.@k$ = .rotation$[.slot];
	if (.@k$ == "TOWN")
		callsub S_StartTownInvasion;
	else if (.@k$ == "BOSS")
		callsub S_StartRoamingBoss;
	end;

// ---- GM overrides (@event, or donpcevent) -----------------
OnStartTown:
	callsub S_StartTownInvasion;
	end;

OnStartBoss:
	callsub S_StartRoamingBoss;
	end;

OnStopEvent:
	if (.running) {
		stopnpctimer;
		killmonster .inv_active_map$, "seroja_events::OnInvKill";
		.running = 0;
	}
	if (.boss_running) {
		killmonster .boss_active_map$, "seroja_events::OnBossKill";
		.boss_running = 0;
	}
	announce "[SeROja] A GM has cleared the running event.", bc_all;
	end;

OnGMEvent:
	mes "[SeROja Events]";
	mes "Trigger an event manually.";
	next;
	switch (select("Start Town Invasion:Start Roaming Boss:Stop / clear:Cancel")) {
	case 1: donpcevent "seroja_events::OnStartTown"; break;
	case 2: donpcevent "seroja_events::OnStartBoss"; break;
	case 3: donpcevent "seroja_events::OnStopEvent"; break;
	}
	close;

// ---- Town Invasion ---------------------------------------
S_StartTownInvasion:
	if (.running)
		return;
	.running = 1;
	deletearray $@ev_aid[0], getarraysize($@ev_aid);
	deletearray $@ev_cid[0], getarraysize($@ev_cid);
	deletearray $@ev_name$[0], getarraysize($@ev_name$);
	deletearray $@ev_score[0], getarraysize($@ev_score);

	.@i = rand(getarraysize(.inv_towns$));
	.inv_active_map$ = .inv_towns$[.@i];
	announce "[SeROja] A " + .inv_name$ + " horde is descending on " + .inv_active_map$
		+ "! Defend it — the top " + .inv_topn + " defenders get mailed.", bc_all;
	monster .inv_active_map$, .inv_x[.@i], .inv_y[.@i], .inv_name$ + " Raider",
		.inv_wave_mob, .inv_wave_qty, "seroja_events::OnInvKill";
	monster .inv_active_map$, .inv_x[.@i], .inv_y[.@i], .inv_name$ + " Warlord",
		.inv_boss_mob, 1, "seroja_events::OnInvKill";
	stopnpctimer;
	setnpctimer 0;
	initnpctimer;
	return;

OnInvKill:
	if (getcharid(3) == 0)      // trap / mob-vs-mob kill — no attached player
		end;
	.@aid = getcharid(3);
	.@n = getarraysize($@ev_aid);
	.@idx = -1;
	for (.@i = 0; .@i < .@n; .@i++)
		if ($@ev_aid[.@i] == .@aid) {
			.@idx = .@i;
			break;
		}
	if (.@idx == -1) {
		.@idx = .@n;
		$@ev_aid[.@idx] = .@aid;
		$@ev_cid[.@idx] = getcharid(0);
		$@ev_name$[.@idx] = strcharinfo(0);
	}
	$@ev_score[.@idx] = $@ev_score[.@idx] + 1;
	end;

// 15 minutes (.inv_minutes) after the invasion starts
OnTimer900000:
	if (!.running)
		end;
	stopnpctimer;
	killmonster .inv_active_map$, "seroja_events::OnInvKill";
	callsub S_PayTownTopN;
	.running = 0;
	end;

S_PayTownTopN:
	.@n = getarraysize($@ev_score);
	if (.@n == 0) {
		announce "[SeROja] " + .inv_active_map$ + " held — but nobody scored a kill.", bc_all;
		return;
	}
	// selection sort desc, carrying the parallel arrays. ponytail: O(n^2) on the
	// number of distinct defenders (tens, not thousands) — fine.
	for (.@i = 0; .@i < .@n - 1; .@i++) {
		.@mx = .@i;
		for (.@j = .@i + 1; .@j < .@n; .@j++)
			if ($@ev_score[.@j] > $@ev_score[.@mx])
				.@mx = .@j;
		if (.@mx != .@i) {
			.@ts = $@ev_score[.@i];  $@ev_score[.@i] = $@ev_score[.@mx];  $@ev_score[.@mx] = .@ts;
			.@tc = $@ev_cid[.@i];    $@ev_cid[.@i]   = $@ev_cid[.@mx];    $@ev_cid[.@mx]   = .@tc;
			.@ta = $@ev_aid[.@i];    $@ev_aid[.@i]   = $@ev_aid[.@mx];    $@ev_aid[.@mx]   = .@ta;
			.@tn$ = $@ev_name$[.@i]; $@ev_name$[.@i] = $@ev_name$[.@mx];  $@ev_name$[.@mx] = .@tn$;
		}
	}
	for (.@i = 0; .@i < .@n; .@i++) {
		if ($@ev_score[.@i] <= 0)
			break;
		callfunc "F_SeROjaBoardBump", "invasion", $@ev_aid[.@i], $@ev_name$[.@i], $@ev_score[.@i];
		if (.@i < .inv_topn) {
			.@tier = (.@i == 0) ? 1 : ((.@i <= 2) ? 2 : 3);
			callfunc "F_EventReward", $@ev_cid[.@i], $@ev_aid[.@i], .@tier;
		} else {
			callfunc "F_EventReward", $@ev_cid[.@i], $@ev_aid[.@i], 0;
		}
	}
	announce "[SeROja] " + .inv_active_map$ + " held. Top defender: " + $@ev_name$[0]
		+ " (" + $@ev_score[0] + " kills). Rewards mailed.", bc_all;
	return;

// ---- Roaming Boss --------------------------------------
S_StartRoamingBoss:
	if (.boss_running)
		return;
	.boss_running = 1;
	.@i = rand(getarraysize(.boss_maps$));
	.boss_active_map$ = .boss_maps$[.@i];
	announce "[SeROja] A " + .boss_name$ + " has been sighted in " + .boss_active_map$ + "!", bc_all;
	monster .boss_active_map$, 0, 0, .boss_name$, .boss_mob, 1, "seroja_events::OnBossKill";
	// ponytail: one npc timer slot is taken by the town invasion, so the boss
	// self-despawn rides a parked sleep in this instance instead of a second timer.
	sleep .boss_despawn_s * 1000;
	if (.boss_running) {
		killmonster .boss_active_map$, "seroja_events::OnBossKill";
		.boss_running = 0;
	}
	end;

OnBossKill:
	.boss_running = 0;
	if (getcharid(3) == 0)
		end;
	callfunc "F_EventReward", getcharid(0), getcharid(3), 4;
	announce "[SeROja] " + strcharinfo(0) + " felled the " + .boss_name$
		+ " in " + .boss_active_map$ + "!", bc_all;
	end;
}

//== F_EventReward — the one payout path for every SeROja event =
//= getarg(0) = character id (mail target)
//= getarg(1) = account id  (coin target, online-only)
//= getarg(2) = tier: 0 participation / 1 top1 / 2 top2-3 /
//=             3 top4-6 / 4 boss last-hit
function	script	F_EventReward	{
	.@cid  = getarg(0);
	.@aid  = getarg(1);
	.@tier = getarg(2);

	// tier 0 = participation: a coin if online, a small zeny mail if not.
	if (.@tier == 0) {
		if (attachrid(.@aid, false)) {
			callfunc "F_AddActCoin", 1;
			detachrid;
		} else {
			mail .@cid, "SeROja Events", "SeROja Event Reward",
				"Thanks for taking part in an SeROja event. A small reward is attached.", 20000;
		}
		return;
	}

	// tiers 1-4: tiered zeny + items by mail (offline-safe).
	if (.@tier == 1)      { .@zeny = 500000; setarray .@iid[0],12208,12210,6635; setarray .@iamt[0],3,3,2; }
	else if (.@tier == 2) { .@zeny = 300000; setarray .@iid[0],12208,12210;      setarray .@iamt[0],2,2;   }
	else if (.@tier == 3) { .@zeny = 150000; setarray .@iid[0],12208,12210;      setarray .@iamt[0],1,1;   }
	else                  { .@zeny = 400000; setarray .@iid[0],6240,6241,6635;   setarray .@iamt[0],5,5,3; }

	mail .@cid, "SeROja Events", "SeROja Event Reward",
		"Thanks for taking part in an SeROja event. Your reward is attached. "
		+ "Any Activity / SeROja Coins were added to your account directly (online only).",
		.@zeny, .@iid, .@iamt;

	// coins only if the winner is still online.
	if (attachrid(.@aid, false)) {
		if (.@tier == 1)      { callfunc "F_AddSeROjaCoin", 3; callfunc "F_AddActCoin", 5; }
		else if (.@tier == 2) { callfunc "F_AddSeROjaCoin", 2; callfunc "F_AddActCoin", 3; }
		else if (.@tier == 3) { callfunc "F_AddSeROjaCoin", 1; callfunc "F_AddActCoin", 2; }
		else                  { callfunc "F_AddSeROjaCoin", 5; callfunc "F_AddActCoin", 3; }
		detachrid;
	}
	return;
}

//== F_SeROjaBoardBump — running counter into www_leaderboard_stats (PLAN-019) =
//= Feeds the website event boards (C5/C6). Same idiom + column contract as
//= npc/custom/seroja/leaderboard.txt: PK (kind, account_id), INSERT ... ON
//= DUPLICATE KEY UPDATE. Zero HTTP from the map-server (ADR-0003 / ADR-0008).
//=   getarg(0) = kind ('invasion' | 'mushroom')   -- fixed literals, not input
//=   getarg(1) = account id (numeric)
//=   getarg(2) = char name (escape_sql'd here)
//=   getarg(3) = amount to add to the running score
function	script	F_SeROjaBoardBump	{
	.@kind$ = getarg(0);
	.@aid   = getarg(1);
	.@name$ = escape_sql(getarg(2));
	.@n     = getarg(3);
	if (.@aid <= 0 || .@n <= 0)
		return;
	query_sql("INSERT INTO `www_leaderboard_stats` (kind, account_id, char_name, score) "
		+ "VALUES ('" + .@kind$ + "', " + .@aid + ", '" + .@name$ + "', " + .@n + ") "
		+ "ON DUPLICATE KEY UPDATE score = score + " + .@n + ", char_name = '" + .@name$ + "'");
	return;
}
